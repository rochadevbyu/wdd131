function definirInformacoesCurso(curso) { document.querySelector("#nomeCurso").innerHTML = `${curso.codigo} – ${curso.nome}`; } function exibirSecoes(curso) { const tbody = document.querySelector("#secoes tbody"); let linhas = ""; for (const secao of curso.secoes) { linhas += ` ${secao.secao} ${secao.matriculado} ${secao.instrutor} `; } tbody.innerHTML = linhas; }
